function escapeHtml(
    value

        .replaceAll(
            "&",

            "<",

        )
        .replaceAll(
            ">",

function renderInlineMarkdown(
    value
) {

    let result =
        escapeHtml(
            value
        );


    result =
        result.replace(
            /`([^`]+)`/g,
            "<code>$1</code>"
        );


    result =
        result.replace(
            /\*\*([^*]+)\*\*/g,
            "<strong>$1</strong>"
        );


    result =
        result.replace(
            /\*([^*]+)\*/g,
            "<em>$1</em>"
        );


    return result;

}


function renderMarkdown(
    content
) {

    if (!content) {

        return {
            __html: ""
        };

    }


    const text =
        String(content)
            .replaceAll(
                "\r\n",
                "\n"
            );


    const lines =
        text.split(
            "\n"
        );


    const output = [];


    let inCodeBlock =
        false;


    let codeBuffer = [];


    let inTable =
        false;


    let tableRows = [];


    function flushTable() {

        if (
            !tableRows.length
        ) {

            return;

        }


        const rows =
            tableRows.filter(
                row =>
                    row.trim() !== ""
            );


        if (!rows.length) {

            tableRows = [];

            inTable = false;

            return;

        }


        let html =
            '<div class="md-table-wrap">' +
            '<table class="md-table">';


        let visibleRowIndex =
            0;


        rows.forEach(
            row => {

                const cells =
                    row
                        .trim()
                        .replace(
                            /^\|/,
                            ""
                        )
                        .replace(
                            /\|$/,
                            ""
                        )
                        .split("|")
                        .map(
                            cell =>
                                cell.trim()
                        );


                const separator =
                    cells.length > 0 &&
                    cells.every(
                        cell =>
                            /^:?-{3,}:?$/.test(
                                cell
                            )
                    );


                if (
                    separator
                ) {

                    return;

                }


                const tag =
                    visibleRowIndex === 0
                        ? "th"
                        : "td";


                html +=
                    "<tr>";


                cells.forEach(
                    cell => {

                        html +=
                            `<${tag}>` +
                            renderInlineMarkdown(
                                cell
                            ) +
                            `</${tag}>`;

                    }
                );


                html +=
                    "</tr>";


                visibleRowIndex++;

            }
        );


        html +=
            "</table></div>";


        output.push(
            html
        );


        tableRows = [];

        inTable = false;

    }


    function flushCode() {

        if (
            !codeBuffer.length
        ) {

            return;

        }


        output.push(
            `<pre class="md-code"><code>${escapeHtml(
                codeBuffer.join("\n")
            )}</code></pre>`
        );


        codeBuffer = [];

    }


    lines.forEach(
        line => {

            if (
                line
                    .trim()
                    .startsWith(
                        "```"
                    )
            ) {

                flushTable();


                if (
                    inCodeBlock
                ) {

                    inCodeBlock =
                        false;

                    flushCode();

                }
                else {

                    inCodeBlock =
                        true;

                }


                return;

            }


            if (
                inCodeBlock
            ) {

                codeBuffer.push(
                    line
                );

                return;

            }


            const trimmed =
                line.trim();


            const looksLikeTable =
                trimmed.startsWith(
                    "|"
                ) &&
                trimmed.endsWith(
                    "|"
                );


            if (
                looksLikeTable
            ) {

                inTable =
                    true;

                tableRows.push(
                    trimmed
                );

                return;

            }


            if (
                inTable
            ) {

                flushTable();

            }


            if (!trimmed) {

                output.push(
                    '<div class="md-space"></div>'
                );

                return;

            }


            if (
                /^###\s+/.test(
                    trimmed
                )
            ) {

                output.push(
                    `<h4>${renderInlineMarkdown(
                        trimmed.replace(
                            /^###\s+/,
                            ""
                        )
                    )}</h4>`
                );

                return;

            }


            if (
                /^##\s+/.test(
                    trimmed
                )
            ) {

                output.push(
                    `<h3>${renderInlineMarkdown(
                        trimmed.replace(
                            /^##\s+/,
                            ""
                        )
                    )}</h3>`
                );

                return;

            }


            if (
                /^#\s+/.test(
                    trimmed
                )
            ) {

                output.push(
                    `<h2>${renderInlineMarkdown(
                        trimmed.replace(
                            /^#\s+/,
                            ""
                        )
                    )}</h2>`
                );

                return;

            }


            if (
                /^[-*]\s+/.test(
                    trimmed
                )
            ) {

                output.push(
                    `<div class="md-list-item">` +
                    `<span>â€¢</span>` +
                    `<div>${renderInlineMarkdown(
                        trimmed.replace(
                            /^[-*]\s+/,
                            ""
                        )
                    )}</div>` +
                    `</div>`
                );

                return;

            }


            if (
                /^\d+\.\s+/.test(
                    trimmed
                )
            ) {

                const number =
                    trimmed.match(
                        /^\d+/
                    )?.[0] ||
                    "";


                output.push(
                    `<div class="md-list-item">` +
                    `<span>${escapeHtml(
                        number
                    )}.</span>` +
                    `<div>${renderInlineMarkdown(
                        trimmed.replace(
                            /^\d+\.\s+/,
                            ""
                        )
                    )}</div>` +
                    `</div>`
                );

                return;

            }


            output.push(
                `<p>${renderInlineMarkdown(
                    trimmed
                )}</p>`
            );

        }
    );


    if (
        inTable
    ) {

        flushTable();

    }


    if (
        inCodeBlock
    ) {

        flushCode();

    }


    return {

        __html:
            output.join("")

    };

}

function hexToRgba(
    hex,
    alpha
) {

    const clean =
        String(
            hex || ""
        )
            .replace(
                "#",
                ""
            );


    if (
        clean.length !== 6
    ) {

        return hex;

    }


    const r =
        parseInt(
            clean.substring(
                0,
                2
            ),
            16
        );


    const g =
        parseInt(
            clean.substring(
                2,
                4
            ),
            16
        );


    const b =
        parseInt(
            clean.substring(
                4,
                6
            ),
            16
        );


    return (
        `rgba(${r}, ${g}, ${b}, ${alpha})`
    );

}

function extractChartPayload(
    message
) {

    if (
        !message
    ) {

        return null;

    }


    const payload =
        message.chart_payload;


    if (
        !payload ||
        !Array.isArray(
            payload.data
        ) ||
        payload.data.length === 0
    ) {

        return null;

    }


    const data =
        payload.data
            .map(
                item => {

                    const label =
                        String(
                            item?.label ??
                            ""
                        );


                    const value =
                        Number(
                            item?.value
                        );


                    return {

                        ...item,

                        label,

                        value

                    };

                }
            )
            .filter(
                item =>
                    item.label &&
                    Number.isFinite(
                        item.value
                    )
            );


    if (
        !data.length
    ) {

        return null;

    }


    /*
     * IMPORTANTE:
     *
     * I payload storici creati prima del Chart Analyzer
     * possono non avere chart_type.
     *
     * In quel caso NON dobbiamo assumere "bar".
     * Recuperiamo il tipo dalla richiesta originale.
     *
     * Esempio:
     *
     * "Mostrami l'andamento degli ordini..."
     *
     * -> line
     *
     * "Fammi un grafico..."
     *
     * -> bar
     *
     * Questo mantiene corretti anche i grafici giÃ 
     * salvati nel database prima della correzione.
     */

    const originalRequest =
        payload.request ||
        message.chart_request ||
        "";


    const resolvedChartType =
        payload.chart_type ||
        getLegacyChartType(
            originalRequest
        );


    return {

        ...payload,

        chart_type:
            resolvedChartType,

        data

    };

}

function getLegacyChartType(
    question
) {

    const text =
        String(
            question || ""
        )
            .toLowerCase();


    /*
     * Ordine intenzionale:
     *
     * 1. torta
     * 2. ciambella
     * 3. linee / andamento
     * 4. bar come fallback
     */


    if (
        text.includes(
            "torta"
        ) ||
        text.includes(

        )
    ) {

    }


    if (
        text.includes(

        )
    ) {

        return "doughnut";

    if (
        text.includes(
            "linea"
        ) ||
        text.includes(
            "linee"
        ) ||
        text.includes(
            "andamento"
        )
    ) {

        return "line";

    }


    return "bar";

}

                        item.value
                    )
            );


    if (
        !data.length
    ) {

        return null;

    }


    const request =
        result.request ||
        message.chart_request ||
        "";


    const legacyChartType =
        getLegacyChartType(
            request
        );


    return {

        request,

        chart_type:
            legacyChartType,

        orientation:
            data.length >= 5
                ? "horizontal"
                : "vertical",

        title:
            request ||
            "Analisi dati",

        grouping_field:
            groupingField,

        value_field:
            valueField,

        sort:
            "none",

        style: {

            palette:
                "professional",

            show_legend:
                legacyChartType ===
                    "pie" ||
                legacyChartType ===
                    "doughnut",

            show_values:
                true,

            animation_preset:
                "dynamic"

        },

        data

    };

}

    /* ======================================================

       ====================================================== */

    async function handleRenameConversation(
        conversation
    ) {

        if (
            loading ||
            !conversation
        ) {

            return;

        }


        const title =
            window.prompt(
                "Nome della conversazione:",
                conversation.title ||
                "Nuova conversazione"
            );


        if (
            title === null
        ) {

            return;

        }


        const clean =
            title.trim();


        if (
            !clean
        ) {

            return;

        }


        try {

            const result =
                await renameConversation(
                    conversation.id,
                    clean
                );


            const updated =
                result?.data;


            if (
                !updated
            ) {

                return;

            }


            updateConversationLocal(
                conversation.id,
                current => ({

                    ...current,

                    ...updated

                })
            );

        }
        catch (error) {

            console.error(
                "Errore rinomina:",
                error
            );

        }

    }


    /* ======================================================
       DELETE
       ====================================================== */

    async function handleDeleteConversation(
        conversation
    ) {

        if (
            loading ||
            !conversation
        ) {

            return;

        }


        const confirmed =
            window.confirm(
                `Eliminare "${conversation.title}"?`
            );


        if (
            !confirmed
        ) {

            return;

        }


        try {

            await deleteConversation(
                conversation.id
            );


            const remaining =
                conversations.filter(
                    current =>
                        current.id !==
                        conversation.id
                );


            if (
                remaining.length ===
                0
            ) {

                const created =
                    await createConversation();


                const fresh = {

                    ...created.data,

                    message_count:
                        created.data?.message_count ??
                        0,

                    messages:
                        []

                };


                setConversations(
                    [
                        fresh
                    ]
                );


                setActiveConversationId(
                    fresh.id
                );

            }
            else {

                setConversations(
                    remaining
                );


                if (
                    conversation.id ===
                    activeConversationId
                ) {

                    setActiveConversationId(
                        remaining[0].id
                    );

                }

            }

        }
        catch (error) {

            console.error(
                error
            );

        }

    }


    /* ======================================================
       SEND
       ====================================================== */

    async function send(
        questionOverride = null
    ) {

        const question =
            (
                questionOverride ??
                input
            )
                .trim();


        if (
            !question ||
            loading ||
            !activeConversation
        ) {

            return;

        }


        const conversationId =
            activeConversation.id;


        const userMessage = {

            id:
                makeLocalId(
                    "msg"
                ),

            role:
                "user",

            content:
                question,

            created_at:
                new Date().toISOString()

        };


        let baseMessages =
            [
                ...(
                    activeConversation.messages ||
                    []
                )
            ];


        if (
            editingMessageId
        ) {

            const editedIndex =
                baseMessages.findIndex(
                    message =>
                        message.id ===
                        editingMessageId
                );


            if (
                editedIndex !== -1
            ) {

                baseMessages =
                    baseMessages.slice(
                        0,
                        editedIndex
                    );

            }

        }


        const optimisticMessages =
            [
                ...baseMessages,
                userMessage
            ];


        updateConversationLocal(
            conversationId,
            conversation => ({

                ...conversation,

                title:
                    optimisticMessages.length ===
                    1
                        ? makeConversationTitle(
                            question
                        )
                        : conversation.title,

                messages:
                    optimisticMessages,

                updated_at:
                    new Date().toISOString()

            })
        );


        setInput(
            ""
        );


        setEditingMessageId(
            null
        );


        setLoading(
            true
        );


        requestAnimationFrame(
            () => {

                if (
                    textareaRef.current
                ) {

                    textareaRef.current.style.height =
                        "auto";

                }

            }
        );


        try {

            const result =
                await sendMessage(
                    question,
                    conversationId
                );


            const response =
                result?.data?.final_response ||
                "Nessuna risposta disponibile.";


            const assistantMessage = {

                id:
                    makeLocalId(
                        "msg"
                    ),

                role:
                    "assistant",

                content:
                    response,

                created_at:
                    new Date().toISOString(),

                chart_request:
                    question,

                odoo_results:
                    result?.data?.odoo_results ||
                    [],

                chart_payload:
                    result?.data?.chart_payload ||
                    null

            };


            const returnedConversationId =
                result?.data?.conversation_id;


            if (
                returnedConversationId &&
                returnedConversationId !==
                    conversationId
            ) {

                console.warn(
                    "Il backend ha restituito una conversation_id diversa:",
                    returnedConversationId
                );

            }


            updateConversationLocal(
                conversationId,
                conversation => ({

                    ...conversation,

                    messages:
                        [
                            ...optimisticMessages,
                            assistantMessage
                        ],

                    message_count:
                        (
                            conversation.message_count ??
                            conversation.messages?.length ??
                            0
                        ) + 2,

                    updated_at:
                        new Date().toISOString()

                })
            );


            if (
                optimisticMessages.length ===
                1
            ) {

                try {

                    const renamed =
                        await renameConversation(
                            conversationId,
                            makeConversationTitle(
                                question
                            )
                        );


                    if (
                        renamed?.data
                    ) {

                        updateConversationLocal(
                            conversationId,
                            current => ({

                                ...current,

                                ...renamed.data,

                                message_count:
                                    current.message_count ??
                                    2

                            })
                        );

                    }

                }
                catch (
                    renameError
                ) {

                    console.warn(
                        "Titolo non aggiornato:",
                        renameError
                    );

                }

            }

        }
        catch (error) {

            const errorMessage = {

                id:
                    makeLocalId(
                        "msg"
                    ),

                role:
                    "error",

                content:
                    error?.message ||
                    "Errore di comunicazione con il backend.",

                created_at:
                    new Date().toISOString()

            };


            updateConversationLocal(
                conversationId,
                conversation => ({

                    ...conversation,

                    messages:
                        [
                            ...optimisticMessages,
                            errorMessage
                        ],

                    updated_at:
                        new Date().toISOString()

                })
            );

        }
        finally {

            setLoading(
                false
            );

        }

    }

    /* ======================================================
       REGENERATE
       ====================================================== */

    async function regenerate(
        message
    ) {

        if (
            loading ||
            !activeConversation ||
            message.role !==
                "assistant"
        ) {

            return;

        }


        const assistantIndex =
            activeConversation.messages.findIndex(
                current =>
                    current.id ===
                    message.id
            );


        if (
            assistantIndex <= 0
        ) {

            return;

        }


        const previousUser =
            activeConversation.messages[
                assistantIndex - 1
            ];


        if (
            previousUser?.role !==
            "user"
        ) {

            return;

        }


        const conversationId =
            activeConversation.id;


        const truncated =
            activeConversation.messages.slice(
                0,
                assistantIndex
            );


        updateConversationLocal(
            conversationId,
            conversation => ({

                ...conversation,

                messages:
                    truncated

            })
        );


        setLoading(
            true
        );


        try {

            const result =
                await sendMessage(
                    previousUser.content,
                    conversationId
                );


            const response =
                result?.data?.final_response ||
                "Nessuna risposta disponibile.";


            const assistantMessage = {

                id:
                    makeLocalId(
                        "msg"
                    ),

                role:
                    "assistant",

                content:
                    response,

                created_at:
                    new Date().toISOString(),

                chart_request:
                    previousUser.content,

                odoo_results:
                    result?.data?.odoo_results ||
                    [],

                chart_payload:
                    result?.data?.chart_payload ||
                    null

            };


            updateConversationLocal(
                conversationId,
                conversation => ({

                    ...conversation,

                    messages:
                        [
                            ...truncated,
                            assistantMessage
                        ],

                    updated_at:
                        new Date().toISOString()

                })
            );

        }
        catch (error) {

            updateConversationLocal(
                conversationId,
                conversation => ({

                    ...conversation,

                    messages:
                        [
                            ...truncated,

                            {

                                id:
                                    makeLocalId(
                                        "msg"
                                    ),

                                role:
                                    "error",

                                content:
                                    error?.message ||
                                    "Errore di comunicazione con il backend.",

                                created_at:
                                    new Date().toISOString()

                            }

                        ]

                })
            );

        }
        finally {

            setLoading(
                false
            );

        }

    }

function formatNumber(
    value
) {

    if (
        typeof value !== "number"
    ) {

        return String(
            value ?? ""
        );

    }

    return new Intl.NumberFormat(
        "it-IT",
        {
            maximumFractionDigits: 2
        }
    ).format(
        value
    );

}


/* ==========================================================

function Chart({
    chart
}) {

    if (
        !chart ||
        !Array.isArray(
            chart.data
        ) ||
        chart.data.length === 0
    ) {

        return null;

    }


    const validData =
        chart.data.filter(
            item =>
                item &&
                item.label !== undefined &&
                typeof item.value === "number"
        );


    if (
        validData.length === 0
    ) {

        return null;

import { useState } from "react";

function LoginForm({ onLogin }) {
    const [email, setEmail] = useState("");
    const [password, setPassword] = useState("");
    const [database, setDatabase] = useState("");
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState("");

    async function handleSubmit(event) {
        event.preventDefault();

        setError("");

        if (!email.trim() || !password.trim() || !database.trim()) {
            setError("Compila tutti i campi richiesti.");
            return;
        }

        try {
            setLoading(true);

            const result = await login({
                email: email.trim(),
                password,
                database: database.trim()
            });

            if (!result?.success) {
                throw new Error(
                    result?.error?.message ||
                    "Autenticazione non riuscita."
                );
            }

            if (result.data?.session_id) {
                localStorage.setItem(
                    "odoo_session",
                    result.data.session_id
                );
            }

            if (result.data?.database) {
                localStorage.setItem(
                    "odoo_database",
                    result.data.database
                );
            } else {
                localStorage.setItem(
                    "odoo_database",
                    database.trim()
                );
            }

            window.dispatchEvent(
                new Event("auth-changed")
            );

            onLogin();
        } catch (err) {
            setError(
                err?.message ||
                "Errore durante il login."
            );
        } finally {
            setLoading(false);
        }
    }

    return (
        <form
            className="login-form"
            onSubmit={handleSubmit}
        >
            <div className="form-group">
                <label>Email</label>

                <input
                    type="email"
                    value={email}
                    onChange={(event) =>
                        setEmail(event.target.value)
                    }
                    placeholder="nome@azienda.it"
                    autoComplete="username"
                />
            </div>

            <div className="form-group">
                <label>Password</label>

                <input
                    type="password"
                    value={password}
                    onChange={(event) =>
                        setPassword(event.target.value)
                    }
                    placeholder="â€¢â€¢â€¢â€¢â€¢â€¢â€¢â€¢"
                    autoComplete="current-password"
                />
            </div>

            <div className="form-group">
                <label>Database</label>

                <input
                    type="text"
                    value={database}
                    onChange={(event) =>
                        setDatabase(event.target.value)
                    }
                    placeholder="Nome database Portant"
                />
            </div>

            {error && (
                <div className="login-error">
                    {error}
                </div>
            )}

            <button
                className="primary-button login-button"
                type="submit"
                disabled={loading}
            >

                <span>â†</span>

const API_BASE =
    "/api";


async function parseResponse(
    response
) {

    let data = null;

    try {

        data =
            await response.json();

    }
    catch {

        throw new Error(
            "Risposta non valida dal backend."
        );

    }


    if (!response.ok) {

        throw new Error(
            data?.error?.message ||
            data?.detail ||
            "Errore HTTP."
        );

    }


    return data;

}
